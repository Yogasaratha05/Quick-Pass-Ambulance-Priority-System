#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
#include <SPI.h>
#include <MFRC522.h>
#include <LiquidCrystal_I2C.h>

// RFID Pins
#define SS_PIN D4    // GPIO2 on ESP8266
#define RST_PIN D3   // GPIO0 on ESP8266
MFRC522 mfrc522(SS_PIN, RST_PIN);

// LCD Display (I2C)
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Buzzer
#define BUZZER_PIN D2  // GPIO4 on ESP8266

// WiFi Credentials
const char* ssid = "Your_WiFi_SSID";
const char* password = "Your_WiFi_Password";

// Web Server
ESP8266WebServer server(80);

// Variables
String currentZone = "NORMAL";
String ambulanceID = "";
bool ambulanceDetected = false;
unsigned long lastRFIDCheck = 0;
const unsigned long RFID_CHECK_INTERVAL = 2000;

// Zone data structure
struct ZoneInfo {
  String name;
  bool buzzerActive;
  String message;
};

ZoneInfo zones[] = {
  {"NORMAL", false, "Clear Route"},
  {"SCHOOL_ZONE", true, "SLOW-SCHOOL ZONE"},
  {"ACCIDENT_ZONE", true, "CAUTION-ACCIDENT"},
  {"TRAFFIC_HEAVY", true, "HEAVY TRAFFIC"}
};

void setup() {
  Serial.begin(115200);
  
  // Initialize components
  initializeComponents();
  
  // Connect to WiFi
  connectToWiFi();
  
  // Setup web server routes
  setupWebServer();
  
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("System Ready");
  lcd.setCursor(0, 1);
  lcd.print("Waiting for RFID");
}

void initializeComponents() {
  // Initialize SPI and RFID
  SPI.begin();
  mfrc522.PCD_Init();
  
  // Initialize LCD
  lcd.init();
  lcd.backlight();
  
  // Initialize Buzzer
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);
  
  Serial.println("All components initialized");
}

void connectToWiFi() {
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Connecting WiFi");
  
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 20) {
    delay(500);
    Serial.print(".");
    lcd.setCursor(attempts % 16, 1);
    lcd.print(".");
    attempts++;
  }
  
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nConnected to WiFi!");
    Serial.print("IP Address: ");
    Serial.println(WiFi.localIP());
    
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("WiFi Connected");
    lcd.setCursor(0, 1);
    lcd.print("IP:");
    lcd.print(WiFi.localIP().toString());
  } else {
    Serial.println("\nFailed to connect to WiFi");
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("WiFi Failed");
  }
  delay(2000);
}

void setupWebServer() {
  server.on("/", HTTP_GET, handleRoot);
  server.on("/status", HTTP_GET, handleStatus);
  server.on("/setzone", HTTP_GET, handleSetZone);
  server.on("/data", HTTP_GET, handleData);
  
  server.begin();
  Serial.println("HTTP server started");
}

void loop() {
  server.handleClient();
  
  // Check RFID periodically
  if (millis() - lastRFIDCheck > RFID_CHECK_INTERVAL) {
    checkRFID();
    lastRFIDCheck = millis();
  }
  
  // Update display
  updateDisplay();
  
  // Handle buzzer based on zone
  handleBuzzer();
}

void checkRFID() {
  // Look for new cards
  if (!mfrc522.PICC_IsNewCardPresent()) {
    return;
  }
  
  if (!mfrc522.PICC_ReadCardSerial()) {
    return;
  }
  
  // Read RFID card
  String content = "";
  for (byte i = 0; i < mfrc522.uid.size; i++) {
    content.concat(String(mfrc522.uid.uidByte[i] < 0x10 ? " 0" : " "));
    content.concat(String(mfrc522.uid.uidByte[i], HEX));
  }
  content.toUpperCase();
  
  ambulanceID = content.substring(1);
  ambulanceDetected = true;
  
  Serial.print("Ambulance Detected - ID: ");
  Serial.println(ambulanceID);
  
  // Visual and audio feedback
  digitalWrite(BUZZER_PIN, HIGH);
  delay(200);
  digitalWrite(BUZZER_PIN, LOW);
  
  mfrc522.PICC_HaltA();
}

void updateDisplay() {
  lcd.clear();
  
  if (ambulanceDetected) {
    // Line 1: Ambulance info
    lcd.setCursor(0, 0);
    lcd.print("AMB:");
    
    // Display first 8 characters of ID
    String displayID = ambulanceID.substring(0, 8);
    lcd.print(displayID);
    
    // Line 2: Zone info
    lcd.setCursor(0, 1);
    ZoneInfo currentZoneInfo = getZoneInfo(currentZone);
    
    // Truncate message if too long for LCD
    String displayMessage = currentZoneInfo.message;
    if (displayMessage.length() > 16) {
      displayMessage = displayMessage.substring(0, 16);
    }
    lcd.print(displayMessage);
  } else {
    lcd.setCursor(0, 0);
    lcd.print("Scan Ambulance");
    lcd.setCursor(0, 1);
    lcd.print("RFID Tag");
  }
}

void handleBuzzer() {
  ZoneInfo zoneInfo = getZoneInfo(currentZone);
  
  if (zoneInfo.buzzerActive && ambulanceDetected) {
    // Beep pattern for alert zones
    static unsigned long lastBuzzerTime = 0;
    static bool buzzerState = false;
    
    if (millis() - lastBuzzerTime > 500) {
      buzzerState = !buzzerState;
      digitalWrite(BUZZER_PIN, buzzerState);
      lastBuzzerTime = millis();
    }
  } else {
    digitalWrite(BUZZER_PIN, LOW);
  }
}

ZoneInfo getZoneInfo(String zoneName) {
  for (int i = 0; i < 4; i++) {
    if (zones[i].name == zoneName) {
      return zones[i];
    }
  }
  return zones[0]; // Return NORMAL if not found
}

// Web Server Handlers
void handleRoot() {
  String html = R"rawliteral(
  <!DOCTYPE html>
  <html>
  <head>
    <title>Ambulance Traffic Control</title>
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <style>
      body { font-family: Arial; margin: 20px; background: #f0f0f0; }
      .container { max-width: 800px; margin: 0 auto; background: white; padding: 20px; border-radius: 10px; }
      .card { background: #e8f4fd; padding: 15px; margin: 10px 0; border-radius: 5px; border-left: 4px solid #2196F3; }
      .zone-btn { padding: 10px 20px; margin: 5px; border: none; border-radius: 5px; cursor: pointer; font-size: 16px; }
      .normal { background: #4CAF50; color: white; }
      .school { background: #FF9800; color: white; }
      .accident { background: #F44336; color: white; }
      .traffic { background: #9C27B0; color: white; }
      .status { padding: 10px; margin: 10px 0; border-radius: 5px; text-align: center; font-weight: bold; }
    </style>
    <script>
      function setZone(zone) {
        fetch('/setzone?zone=' + zone)
          .then(response => response.json())
          .then(data => {
            updateStatus();
          });
      }
      
      function updateStatus() {
        fetch('/data')
          .then(response => response.json())
          .then(data => {
            document.getElementById('ambulanceStatus').innerHTML = 
              data.ambulanceDetected ? 
              'Ambulance Detected - ID: ' + data.ambulanceID : 
              'No Ambulance Detected';
            document.getElementById('zoneStatus').innerHTML = 
              'Current Zone: ' + data.currentZone;
            document.getElementById('zoneMessage').innerHTML = 
              'Message: ' + data.zoneMessage;
            document.getElementById('buzzerStatus').innerHTML = 
              'Buzzer: ' + (data.buzzerActive ? 'ACTIVE' : 'INACTIVE');
            
            // Update status color
            const statusDiv = document.getElementById('systemStatus');
            statusDiv.style.background = data.ambulanceDetected ? '#FFEB3B' : '#C8E6C9';
          });
      }
      
      setInterval(updateStatus, 2000);
      window.onload = updateStatus;
    </script>
  </head>
  <body>
    <div class="container">
      <h1>🚑 Ambulance Traffic Control System</h1>
      
      <div class="card">
        <h2>System Status</h2>
        <div id="systemStatus" class="status">
          <div id="ambulanceStatus">Loading...</div>
          <div id="zoneStatus">Loading...</div>
          <div id="zoneMessage">Loading...</div>
          <div id="buzzerStatus">Loading...</div>
        </div>
      </div>
      
      <div class="card">
        <h2>Zone Control</h2>
        <button class="zone-btn normal" onclick="setZone('NORMAL')">Normal Zone</button>
        <button class="zone-btn school" onclick="setZone('SCHOOL_ZONE')">School Zone</button>
        <button class="zone-btn accident" onclick="setZone('ACCIDENT_ZONE')">Accident Zone</button>
        <button class="zone-btn traffic" onclick="setZone('TRAFFIC_HEAVY')">Heavy Traffic</button>
      </div>
      
      <div class="card">
        <h2>Instructions</h2>
        <p>1. Scan ambulance RFID tag on the reader</p>
        <p>2. Select zone type from buttons above</p>
        <p>3. LCD will show current status and alerts</p>
        <p>4. Buzzer will activate in alert zones</p>
      </div>
    </div>
  </body>
  </html>
  )rawliteral";
  
  server.send(200, "text/html", html);
}

void handleStatus() {
  String json = "{";
  json += "\"ambulanceDetected\":" + String(ambulanceDetected ? "true" : "false") + ",";
  json += "\"ambulanceID\":\"" + ambulanceID + "\",";
  json += "\"currentZone\":\"" + currentZone + "\",";
  json += "\"zoneMessage\":\"" + getZoneInfo(currentZone).message + "\",";
  json += "\"buzzerActive\":" + String(getZoneInfo(currentZone).buzzerActive ? "true" : "false");
  json += "}";
  
  server.send(200, "application/json", json);
}

void handleSetZone() {
  if (server.hasArg("zone")) {
    currentZone = server.arg("zone");
    Serial.println("Zone changed to: " + currentZone);
    
    // Send JSON response
    String json = "{\"status\":\"success\",\"zone\":\"" + currentZone + "\"}";
    server.send(200, "application/json", json);
  } else {
    server.send(400, "application/json", "{\"status\":\"error\",\"message\":\"No zone specified\"}");
  }
}

void handleData() {
  handleStatus(); // Reuse status handler for data endpoint
}
